cmake_minimum_required(VERSION 3.20)
project(capture_card VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OBS REQUIRED IMPORTED_TARGET libobs)
pkg_check_modules(FRONTEND REQUIRED IMPORTED_TARGET obs-frontend-api)
# libobs.pc embeds a C-only standard flag; CMake sets standards per language.
foreach(TARGET IN ITEMS PkgConfig::OBS PkgConfig::FRONTEND)
  get_target_property(OBS_OPTIONS ${TARGET} INTERFACE_COMPILE_OPTIONS)
  list(REMOVE_ITEM OBS_OPTIONS "-std=gnu17")
  set_property(TARGET ${TARGET} PROPERTY INTERFACE_COMPILE_OPTIONS "${OBS_OPTIONS}")
endforeach()
find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_path(OBS_HEADERS obs-module.h HINTS ${OBS_INCLUDE_DIRS} PATH_SUFFIXES obs REQUIRED)
find_package(Threads REQUIRED)
add_library(capture-card MODULE src/capture-card.c src/frame-health.c src/toolbar.cpp)
target_link_libraries(capture-card PRIVATE PkgConfig::OBS PkgConfig::FRONTEND Qt6::Widgets Threads::Threads)
target_include_directories(capture-card PRIVATE ${OBS_HEADERS})
target_compile_options(capture-card PRIVATE -Wall -Wextra -Werror)
set_target_properties(capture-card PROPERTIES PREFIX "")
include(CTest)
if(BUILD_TESTING)
  add_executable(capture-card-health-test tests/frame-health.c)
  target_compile_options(capture-card-health-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME frame-health COMMAND capture-card-health-test)
  add_executable(capture-card-test tests/lifecycle.c)
  target_link_libraries(capture-card-test PRIVATE PkgConfig::OBS Threads::Threads)
  target_include_directories(capture-card-test PRIVATE ${OBS_HEADERS})
  target_compile_options(capture-card-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME lifecycle COMMAND capture-card-test $<TARGET_FILE:capture-card>)
  set_tests_properties(lifecycle PROPERTIES TIMEOUT 20)
  add_executable(capture-card-toolbar-test tests/toolbar.cpp src/toolbar.cpp)
  target_include_directories(capture-card-toolbar-test PRIVATE ${OBS_HEADERS} src)
  target_link_libraries(capture-card-toolbar-test PRIVATE PkgConfig::OBS PkgConfig::FRONTEND Qt6::Widgets)
  target_compile_options(capture-card-toolbar-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME toolbar COMMAND capture-card-toolbar-test)
  set_tests_properties(toolbar PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11)
  add_executable(capture-card-smoke tests/hardware-smoke.c)
  target_include_directories(capture-card-smoke PRIVATE ${OBS_HEADERS})
  target_link_libraries(capture-card-smoke PRIVATE PkgConfig::OBS PkgConfig::X11 Threads::Threads)
  target_compile_options(capture-card-smoke PRIVATE -Wall -Wextra -Werror)
endif()
